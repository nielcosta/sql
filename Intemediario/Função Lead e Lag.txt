/* Trabalhando com Função Lead e Lag */
with vendas as (
    select 
        t1.seller_id,
        strftime('%Y-%m-%d', t2.order_approved_at) as data_venda,
        sum(t1.price) as valor_venda
    from tb_order_items as t1
    left join tb_orders as t2
        on t1.order_id = t2.order_id
    where t2.order_approved_at is not null -- Filtra registros sem data de aprovação
    group by 
        t1.seller_id, 
        strftime('%Y-%m-%d', t2.order_approved_at) -- Agrupa pela data formatada (sem horário)
)

select 
    seller_id,
    data_venda,
    valor_venda, 
    coalesce(lag(valor_venda) over (
        partition by seller_id 
        order by data_venda asc
    ), 0) as valor_venda_anterior,
     coalesce(lead(valor_venda) over (
        partition by seller_id 
        order by data_venda asc
    ), 0)  as proxima_venda
from vendas
order by seller_id, data_venda asc; -- Corrigido de 'data_pedido' para 'data_venda'

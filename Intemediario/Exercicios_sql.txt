/* Renomear campos de acordo com sua catergoria */

SELECT 
	distinct
      case 
          when product_category_name like '%construcao%' then 'construcao'
      	  when product_category_name like '%casa%' then 'casa'
          when product_category_name like 'moda%' then 'moda'
          when product_category_name like 'fashion%' then 'fashion'
          when product_category_name like 'livros%' then 'livros'
          when product_category_name like 'alimentos%' then 'alimentos'
          when product_category_name like 'artigos%' then 'artigos'
          when product_category_name like 'eletrodomesticos%' then 'eletrodomesticos'
          when product_category_name like 'moveis%' then 'moveis'
          when product_category_name like 'telefonia%' then 'telefonia'
          when product_category_name IS NULL then 'PRODUTO NÃO CADASTRADO'
          else product_category_name
      end 
      as categorias_tratadas
from tb_products
order by 1

/* Qualificar os reviews pela pontuação*/

SELECT
    review_id,
    review_score,
    CASE
        WHEN review_score <= 2 THEN 'Ruim'
        WHEN review_score = 3 THEN 'Neutro'
        WHEN review_score >= 4 THEN 'Bom'
        ELSE 'Sem Score'
    END AS Qualificação
FROM tb_order_reviews;

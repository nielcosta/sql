SELECT
    t1.seller_id,
    t1.seller_state,
    COUNT(DISTINCT t2.order_id) AS qntd_pedidos,

    ROUND(SUM(t2.price), 1) AS vendas_totais,
    ROUND(SUM(t2.freight_value), 1) AS frete,

    ROUND(
        SUM(t2.price) + SUM(t2.freight_value),
        1
    ) AS total_pedido,

    ROUND(
        SUM(
            CASE
                WHEN t4.payment_type = 'credit_card'
                THEN t4.payment_value
                ELSE 0
            END
        ),
        1
    ) AS credit_card,

    ROUND(
        SUM(
            CASE
                WHEN t4.payment_type = 'boleto'
                THEN t4.payment_value
                ELSE 0
            END
        ),
        1
    ) AS boleto,

    ROUND(
        SUM(
            CASE
                WHEN t4.payment_type = 'voucher'
                THEN t4.payment_value
                ELSE 0
            END
        ),
        1
    ) AS voucher,

    ROUND(
        SUM(
            CASE
                WHEN t4.payment_type = 'debit_card'
                THEN t4.payment_value
                ELSE 0
            END
        ),
        1
    ) AS debit_card

FROM tb_sellers t1
LEFT JOIN tb_order_items t2
    ON t1.seller_id = t2.seller_id
LEFT JOIN tb_order_payments t4
    ON t2.order_id = t4.order_id

WHERE t1.seller_id LIKE '024b564ae893c%'

GROUP BY
    t1.seller_id,
    t1.seller_state;

SELECT
    t1.seller_id,
    t2.product_category_name, 
    count(t3.order_id) as Qtde_Vendida,
    sum(t3.payment_value) as Valor_Vendido
FROM tb_order_items as t1
LEFT join tb_products as t2
on t1.product_id = t2.product_id
left join tb_order_payments as t3
on t3.order_id = t1.order_id
group by seller_id
order by Qtde_Vendida desc limit 1

/*Selecionar as maiores vendas por categoria  de  Produtos.*/

WITH vendas_por_categoria AS (
    SELECT 
        t1.seller_id,
        t2.product_category_name AS categoria_produto,
        COUNT(t1.order_item_id) AS quantidade_vendas,
        SUM(t1.price) AS valor_total_vendas,
        ROW_NUMBER() OVER (
            PARTITION BY t1.seller_id 
            ORDER BY COUNT(t1.order_item_id) DESC, SUM(t1.price) DESC
        ) AS ranking
    FROM tb_order_items AS t1
    LEFT JOIN tb_products AS t2
        ON t1.product_id = t2.product_id
    WHERE t2.product_category_name IS NOT NULL -- Remove categorias ausentes/nulas
    GROUP BY 
        t1.seller_id, 
        t2.product_category_name
)

SELECT 
    seller_id,
    categoria_produto,
    quantidade_vendas,
    valor_total_vendas
FROM vendas_por_categoria
WHERE ranking = 1
ORDER BY valor_total_vendas DESC;

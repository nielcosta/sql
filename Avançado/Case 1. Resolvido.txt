/*Monte uma tabela que mostre o seller_id, a categoria do produto mais vendido, a quantidade de vendas desse produto e o valor total de vendas.*/


With Vendas as (
      select 
        t1.seller_id,
        t1.product_id,
        t2.product_category_name,
        count(DISTINCT(t1.order_id)) AS Qtde_Vendas,
        sum(payment_value) as Valor_Venda
      from tb_order_items as t1

      left join tb_products  as t2 
      on  t1.product_id = t2.product_id

      left join tb_order_payments  as t3
      on  t3.order_id = t1.order_id

      group by t1.seller_id, t1.product_id,t2.product_category_name
      
),

Ranking as (
SELECT 
* ,
ROW_NUMBER() OVER (PARTITION BY seller_id ORDER BY Qtde_Vendas DESC, Valor_Venda DESC) AS Ranking
from vendas
)

select * from Ranking
where Ranking = 1
